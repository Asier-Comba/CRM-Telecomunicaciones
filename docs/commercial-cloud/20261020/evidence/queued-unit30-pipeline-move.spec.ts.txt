import {expect,test} from '@playwright/test'
import {startPipelineCurrentFixture} from '../../tests/e2e-preview/fixtures/pipeline-current-server.cjs'
const customer='10000000-0000-4000-8000-000000000001',id='30000000-0000-4000-8000-000000000001'
const stages=['20000000-0000-4000-8000-000000000001','20000000-0000-4000-8000-000000000002','20000000-0000-4000-8000-000000000003']
const title='Oportunidad sintética de pipeline'
let fixture:Awaited<ReturnType<typeof startPipelineCurrentFixture>>
test.beforeAll(async()=>{fixture=await startPipelineCurrentFixture()})
test.afterAll(async()=>{await fixture?.close()})
for(const mode of ['transport','malformed','wrong_receipt','read_failure','stale','wrong_stage','wrong_status','newer','denied','success'] as const)test(`pipeline move retains ${mode} recovery`,async({page},info)=>{
 const commands:{operation:string;input:Record<string,unknown>}[]=[],effects=new Set<string>(),errors:string[]=[]
 let applied=false,corrected=false,readbacks=0
 page.on('pageerror',e=>errors.push(e.message))
 await page.route('**/api/**',async route=>{
  const q=route.request().postDataJSON();let data:unknown
  if(route.request().url().endsWith('/commands')){
   commands.push(q)
   if(q.operation!=='opportunity.change_stage'||mode==='denied'){await route.fulfill({status:403,json:{ok:false,error:'access_denied'}});return}
   applied=true;effects.add(q.input.command_id)
   if(commands.length===1&&mode==='transport'){await route.abort('failed');return}
   if(commands.length===1&&mode==='malformed'){await route.fulfill({contentType:'application/json',body:'{'});return}
   await route.fulfill({json:{ok:true,receipt:{contract_version:'product.v1',operation:q.operation,command_id:commands.length===1&&mode==='wrong_receipt'?'70000000-0000-4000-8000-000000000009':q.input.command_id,id,version:2,status:'open'}}});return
  }
  if(q.operation==='opportunity.stages')data={contract_version:'product.v1',items:stages.map((key,i)=>({id:key,code:['new','qualified','proposal'][i],display_name:['Nueva sintética','Cualificada sintética','Propuesta sintética'][i],position:i,outcome:null,status:'active'})),next_id:null}
  else if(q.operation==='global.search')data={contract_version:'product.v1',items:[{kind:'customer',id:customer,customer_id:customer,label:'Empresa sintética de pipeline',status:'active'},{kind:'opportunity',id,customer_id:customer,label:title,status:'open'}]}
  else if(q.operation==='work.get'){
   if(applied)readbacks++
   if(applied&&!corrected&&mode==='read_failure'){await route.fulfill({status:500,json:{ok:false,error:'internal_safe'}});return}
   const old=!applied||!corrected&&mode==='stale',wrong=applied&&!corrected&&mode==='wrong_stage',newer=applied&&mode==='newer'
   data={contract_version:'product.v1',kind:'opportunity',record:{id,version:old?1:newer?3:2,title,status:applied&&!corrected&&mode==='wrong_status'?'lost':'open',customer_id:customer,stage_id:old||wrong?stages[0]:newer?stages[2]:stages[1],owner_user_id:null,amount_minor:10000,currency:'EUR',next_follow_up_at:null,expected_close_date:null,next_action:null,source:'manual',links:[],history:[],history_partial:false}}
  }else{await route.fulfill({status:403,json:{ok:false,error:'access_denied'}});return}
  await route.fulfill({json:{ok:true,data}})
 })
 try{
  await page.goto(fixture.url)
  await page.getByRole('textbox',{name:'Consultar pipeline local',exact:true}).fill('sintética')
  await page.getByRole('button',{name:'Consultar pipeline',exact:true}).click()
  const select=page.getByRole('combobox',{name:'Mover etapa de '+title,exact:true}),card=page.locator(`[data-opportunity-id="${id}"]`)
  await expect(select).toBeVisible();await select.selectOption(stages[1])
  await expect.poll(()=>commands.length).toBe(1)
  if(!['newer','denied','success'].includes(mode)){
   const retry=page.getByRole('button',{name:'Comprobar el mismo movimiento',exact:true})
   await expect(retry).toBeEnabled();await expect(select).toHaveCount(0)
   await card.click();await expect(page.getByRole('dialog')).toHaveCount(0)
   const before=readbacks
   await page.screenshot({path:info.outputPath(`pipeline-${mode}-pending.png`)})
   corrected=true;await retry.click()
   if(['transport','malformed','wrong_receipt'].includes(mode)){await expect.poll(()=>commands.length).toBe(2);expect(commands[1]).toEqual(commands[0])}
   else{await expect.poll(()=>readbacks).toBe(before+1);expect(commands).toHaveLength(1)}
  }else if(mode==='denied'){
   await expect(page.getByRole('alert')).toBeVisible();await expect(select).toBeEnabled()
   await expect(page.getByRole('button',{name:'Comprobar el mismo movimiento',exact:true})).toHaveCount(0)
   expect(effects.size).toBe(0)
  }
  await expect(select).toHaveValue(mode==='denied'?stages[0]:mode==='newer'?stages[2]:stages[1])
  if(mode!=='denied'){expect(effects.size).toBe(1);await expect(page.getByRole('button',{name:'Comprobar el mismo movimiento',exact:true})).toHaveCount(0)}
  expect(commands.every(q=>q.operation==='opportunity.change_stage')).toBe(true)
  expect(commands[0].input).toMatchObject({id,expected_version:1,stage_id:stages[1]})
  expect(Object.keys(commands[0].input).sort()).toEqual(['command_id','expected_version','id','stage_id'])
  expect(errors).toEqual([]);expect(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth+1)).toBe(true)
  await page.screenshot({path:info.outputPath(`pipeline-${mode}-result.png`)})
 }finally{await page.close()}
})
